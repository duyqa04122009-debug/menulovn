local Players = game:GetService('Players')
local Http = game:GetService('HttpService')
local Teleport = game:GetService('TeleportService')
local UIS = game:GetService('UserInputService')
local player = Players.LocalPlayer
local ENV = (getgenv and getgenv()) or _G
if ENV.TraiCayCleanup then pcall(ENV.TraiCayCleanup) end
local connections, alive = {}, true
local function connect(signal, fn)
    local c = signal:Connect(fn); table.insert(connections, c); return c
end
local config = {AntiAFK=false, WatchAdmins=false, Reload=false, SourceURL='', AdminIds={}}
local configFile = 'TraiCayConfig.json'
if readfile and isfile and isfile(configFile) then
    local ok, saved = pcall(function() return Http:JSONDecode(readfile(configFile)) end)
    if ok and type(saved)=='table' then
        for k,v in pairs(saved) do if config[k] ~= nil then config[k]=v end end
    end
end
local function save()
    if writefile then pcall(function() writefile(configFile, Http:JSONEncode(config)) end) end
end
local gui = Instance.new('ScreenGui')
gui.Name='TraiCayMenu'; gui.ResetOnSpawn=false
local attached = pcall(function() gui.Parent=gethui and gethui() or game:GetService('CoreGui') end)
if not attached then gui.Parent=player:WaitForChild('PlayerGui') end
local panel=Instance.new('Frame',gui)
panel.Size=UDim2.fromOffset(300,375); panel.Position=UDim2.new(.5,-150,.5,-187)
panel.BackgroundColor3=Color3.fromRGB(24,28,40)
Instance.new('UICorner',panel).CornerRadius=UDim.new(0,12)
local title=Instance.new('TextLabel',panel)
title.Size=UDim2.new(1,0,0,44); title.BackgroundTransparency=1
 title.Text='TRÁI CÂY • BỘ KHUNG'; title.TextColor3=Color3.new(1,1,1); title.TextSize=18
local list=Instance.new('ScrollingFrame',panel)
list.Position=UDim2.fromOffset(10,46); list.Size=UDim2.new(1,-20,1,-92)
list.BackgroundTransparency=1; list.BorderSizePixel=0; list.AutomaticCanvasSize=Enum.AutomaticSize.Y
list.CanvasSize=UDim2.new(); list.ScrollBarThickness=4
local layout=Instance.new('UIListLayout',list); layout.Padding=UDim.new(0,7)
local status=Instance.new('TextLabel',panel)
status.Position=UDim2.new(0,10,1,-43); status.Size=UDim2.new(1,-20,0,38)
status.BackgroundTransparency=1; status.TextWrapped=true; status.TextSize=12
status.TextColor3=Color3.fromRGB(180,200,220); status.Text='Sea hiện tại: PlaceId '..game.PlaceId
local function note(t) if alive then status.Text=t end end
local function button(text, fn)
    local b=Instance.new('TextButton',list); b.Size=UDim2.new(1,-6,0,38)
    b.BackgroundColor3=Color3.fromRGB(43,53,75); b.TextColor3=Color3.new(1,1,1)
    b.TextSize=14; b.Text=text; Instance.new('UICorner',b)
    connect(b.Activated,function() local ok,e=pcall(fn); if not ok then note(tostring(e)) end end)
    return b
end
local function toggle(label,key)
    local b
    local function paint() b.Text=label..': '..(config[key] and 'BẬT' or 'TẮT') end
    b=button('',function() config[key]=not config[key]; save(); paint() end); paint()
end
-- Drag title using mouse or touch.
local drag, origin, initial
connect(title.InputBegan,function(input)
    if input.UserInputType==Enum.UserInputType.Touch or input.UserInputType==Enum.UserInputType.MouseButton1 then
        drag=input; origin=input.Position; initial=panel.Position
    end
end)
connect(UIS.InputChanged,function(input)
    if drag and (input==drag or input.UserInputType==Enum.UserInputType.MouseMovement) then
        local d=input.Position-origin
        panel.Position=UDim2.new(initial.X.Scale,initial.X.Offset+d.X,initial.Y.Scale,initial.Y.Offset+d.Y)
    end
end)
connect(UIS.InputEnded,function(input) if input==drag then drag=nil end end)
local mini=Instance.new('TextButton',gui)
mini.Size=UDim2.fromOffset(56,38); mini.Position=UDim2.new(0,12,.5,-19)
mini.Text='MENU'; mini.BackgroundColor3=Color3.fromRGB(43,53,75); mini.TextColor3=Color3.new(1,1,1)
Instance.new('UICorner',mini)
connect(mini.Activated,function() panel.Visible=not panel.Visible end)
toggle('Anti-AFK (thử nghiệm)','AntiAFK')
connect(player.Idled,function()
    if not config.AntiAFK then return end
    local ok=pcall(function()
        local vu=game:GetService('VirtualUser')
        vu:CaptureController(); vu:ClickButton2(Vector2.new(0,0))
    end)
    if not ok then note('Delta hiện tại không hỗ trợ Anti-AFK này.') end
end)
local hopping=false
local function hop()
    if hopping then return end
    hopping=true; note('Đang tìm server công khai...')
    task.spawn(function()
        local ok,err=pcall(function()
            local cursor=''
            for page=1,5 do
                local url=('https://games.roblox.com/v1/games/%s/servers/Public?sortOrder=Asc&limit=100%s'):format(game.PlaceId,cursor~='' and '&cursor='..Http:UrlEncode(cursor) or '')
                local data=Http:JSONDecode(game:HttpGet(url))
                for _,server in ipairs(data.data or {}) do
                    if server.id~=game.JobId and server.playing<server.maxPlayers then
                        Teleport:TeleportToPlaceInstance(game.PlaceId,server.id,player)
                        note('Đã gửi yêu cầu đổi server.'); return
                    end
                end
                cursor=data.nextPageCursor or ''; if cursor=='' then break end
                task.wait(.5)
            end
            note('Không tìm được server còn chỗ trong 5 trang đầu.')
        end)
        if not ok then note('Đổi server thất bại: '..tostring(err)) end
        hopping=false
    end)
end
button('Đổi server cùng Sea',hop)
connect(Teleport.TeleportInitFailed,function(p,_,message)
    if p==player then hopping=false; note('Teleport thất bại: '..tostring(message)) end
end)
toggle('Theo dõi danh sách admin','WatchAdmins')
local function inspect(p)
    if not config.WatchAdmins or p==player then return end
    for _,id in ipairs(config.AdminIds) do
        if p.UserId==tonumber(id) then note('Gặp UserId trong danh sách theo dõi.'); hop(); return end
    end
end
connect(Players.PlayerAdded,inspect)
-- Recheck existing players periodically, including after enabling the toggle.
task.spawn(function()
    while alive do
        if config.WatchAdmins then for _,p in ipairs(Players:GetPlayers()) do inspect(p) end end
        task.wait(8)
    end
end)
toggle('Tự nạp sau teleport','Reload')
connect(player.OnTeleport,function(state)
    if state~=Enum.TeleportState.Started or not config.Reload then return end
    local queue=queue_on_teleport or queueonteleport or (syn and syn.queue_on_teleport)
    if type(queue)~='function' then note('Không có queue_on_teleport.'); return end
    if not tostring(config.SourceURL):match('^https://') then note('Chưa đặt SourceURL HTTPS của script bạn.'); return end
    queue(('loadstring(game:HttpGet(%q))()'):format(config.SourceURL))
end)

    alive=false
    for _,c in ipairs(connections) do c:Disconnect() end
