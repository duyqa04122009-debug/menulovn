function HasItem(name)
 for _, v in pairs(Backpack:GetChildren()) do
 if v.Name == name then return true end
 end
 return false
end

function EquipWeapon(name)
 local item = Backpack:FindFirstChild(name)
 if item then Humanoid:EquipTool(item) end
end

function GetClosestEnemy()
 local closest, nearest = nil, math.huge
 local myPos = RootPart.Position
 for _, v in pairs(workspace.Enemies:GetChildren()) do -- Đổi từ Players thành workspace.Enemies cho đúng game Blox Fruits
 if v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
 local dist = (myPos - v.HumanoidRootPart.Position).Magnitude
 if dist < nearest then nearest, closest = dist, v end
 end
 end
 return closest
end

function AutoAcceptQuest()
 local Main = LocalPlayer.PlayerGui:FindFirstChild("Main")
 if Main and Main:FindFirstChild("QuestContainer") then
 local QBtn = Main.QuestContainer.QuestDetails.QuestButton
 if QBtn and QBtn:FindFirstChild("AcceptQuest") then
 QBtn.AcceptQuest:Click()
 end
 end
end

-- === TÁC VỤ CHÍNH (LOOPS TỐI ƯU) ===

-- 1. Auto Farm Level & Mastery
spawn(function()
 while task.wait(0.5) do
 if Settings.AutoFarm.Level then AutoAcceptQuest() end
 if Settings.AutoFarm.Level or Settings.AutoFarm.Mastery then
 local target = GetClosestEnemy()
 if target and target:FindFirstChild("HumanoidRootPart") then
 AttackTarget(target)
 local dist = (RootPart.Position - target.HumanoidRootPart.Position).Magnitude
 if dist > 50 then 
    RootPart.CFrame = target.HumanoidRootPart.CFrame * CFrame.new(0, 5, 0) -- Teleport trực tiếp tới quái thay vì đi bộ vòng vèo
 end
 end
 end
 if Settings.AutoFarm.StoreFruit then AutoStoreFruit() end
 end
end)

-- 2. Fast Attack

 end

end)

-- 3. Auto Select Weapon
spawn(function()
 while task.wait(1) do
 if Settings.AutoFarm.AutoSelectWeapon then
 if HasItem("Devil Fruit") then EquipWeapon("Devil Fruit")
 elseif HasItem("Dragon Key") then EquipWeapon("Dragon Key")
 elseif HasItem("Sword") then EquipWeapon("Sword")
 elseif HasItem("Katana") then EquipWeapon("Katana")
 elseif HasItem("Melee") then EquipWeapon("Melee")
 end
 end
 end
end)

-- 4. Sea Events
spawn(function()
 while task.wait(2) do
 if Settings.SeaEvents.AutoEvents then
 for _, v in pairs(workspace:GetChildren()) do
 if v.Name == "SeaEvent" or v.Name == "Event" then
 for _, c in pairs(v:GetChildren()) do
 local boss = c:FindFirstChild("SeaEventBoss") or c:FindFirstChild("Boss")
 if boss and boss:FindFirstChild("HumanoidRootPart") then
 RootPart.CFrame = boss.HumanoidRootPart.CFrame + Vector3.new(0, 10, 0)
 AttackTarget(boss)
 break
 end
 end
 end
 end
 end
 end
end)

-- 5. Player Stats & Movement
spawn(function()
 while task.wait(0.1) do

end)

 end
