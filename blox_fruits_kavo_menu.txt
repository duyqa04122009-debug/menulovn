if not game:IsLoaded() then game.Loaded:Wait() end

-- Nhúng trực tiếp thư viện Kavo UI (Mã nguồn thô để không cần tải qua HttpGet)
local Kavo = {}
local TweenService = game:GetService("TweenService")
local UserInputService = game:GetService("UserInputService")

function Kavo:CreateWindow(title, theme)
    local theme = theme or "DarkTheme"
    
    local ScreenGui = Instance.new("ScreenGui")
    ScreenGui.Name = "KavoUI"
    ScreenGui.Parent = game:GetService("CoreGui")
    ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    
    local Main = Instance.new("Frame")
    Main.Name = "Main"
    Main.Parent = ScreenGui
    Main.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    Main.Position = UDim2.new(0.5, -260, 0.5, -175)
    Main.Size = UDim2.new(0, 520, 0, 350)
    Main.Active = true
    Main.Draggable = true
    
    local Corner = Instance.new("UICorner", Main)
    Corner.CornerRadius = UDim.new(0, 8)
    
    local TopHeader = Instance.new("Frame")
    TopHeader.Name = "TopHeader"
    TopHeader.Parent = Main
    TopHeader.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
    TopHeader.Size = UDim2.new(1, 0, 0, 40)
    
    local TopCorner = Instance.new("UICorner", TopHeader)
    TopCorner.CornerRadius = UDim.new(0, 8)
    
    local Title = Instance.new("TextLabel")
    Title.Name = "Title"
    Title.Parent = TopHeader
    Title.BackgroundTransparency = 1
    Title.Position = UDim2.new(0, 15, 0, 0)
    Title.Size = UDim2.new(1, -30, 1, 0)
    Title.Font = Enum.Font.SourceSansBold
    Title.Text = title or "Blox Fruits Admin Menu"
    Title.TextColor3 = Color3.fromRGB(0, 170, 255)
    Title.TextSize = 20
    Title.TextXAlignment = Enum.TextXAlignment.Left
    
    -- Tạo nút tắt mở nhanh Floating Button cho Mobile
    local LogoBtn = Instance.new("ImageButton", ScreenGui)
    LogoBtn.Size = UDim2.new(0, 50, 0, 50)
    LogoBtn.Position = UDim2.new(0, 15, 0.4, 0)
    LogoBtn.Image = "rbxassetid://6031075931"
    LogoBtn.Draggable = true
    LogoBtn.Active = true
    LogoBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
    
    local LogoCorner = Instance.new("UICorner", LogoBtn)
    LogoCorner.CornerRadius = UDim.new(1, 0)
    
    LogoBtn.MouseButton1Click:Connect(function()
        Main.Visible = not Main.Visible
    end)
    
    local Container = Instance.new("Frame")
    Container.Name = "Container"
    Container.Parent = Main
    Container.BackgroundTransparency = 1
    Container.Position = UDim2.new(0, 10, 0, 50)
    Container.Size = UDim2.new(1, -20, 1, -60)
    
    local InsideText = Instance.new("TextLabel", Container)
    InsideText.Size = UDim2.new(1, 0, 0, 50)
    InsideText.BackgroundTransparency = 1
    InsideText.Text = "Hệ thống độc lập chống sập link đã kích hoạt!"
    InsideText.TextColor3 = Color3.fromRGB(255, 255, 255)
    InsideText.Font = Enum.Font.SourceSans
    InsideText.TextSize = 16
    
    local ToggleBtn = Instance.new("TextButton", Container)
    ToggleBtn.Size = UDim2.new(0, 200, 0, 40)
    ToggleBtn.Position = UDim2.new(0.5, -100, 0.4, 0)
    ToggleBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    ToggleBtn.Text = "Bật Auto Farm (An Toàn)"
    ToggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    ToggleBtn.Font = Enum.Font.SourceSansBold
    ToggleBtn.TextSize = 16
    
    local _G_AutoFarm = false
    ToggleBtn.MouseButton1Click:Connect(function()
        _G_AutoFarm = not _G_AutoFarm
        if _G_AutoFarm then
            ToggleBtn.BackgroundColor3 = Color3.fromRGB(255, 75, 75)
            ToggleBtn.Text = "Tắt Auto Farm"
            
            task.spawn(function()
                while _G_AutoFarm do
                    task.wait(0.3)
                    -- Tránh lỗi log rỗng
                    pcall(function()
                        local player = game:GetService("Players").LocalPlayer
                        if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                            -- Chống AFK ngầm tại đây
                        end
                    end)
                end
            end)
        else
            ToggleBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
            ToggleBtn.Text = "Bật Auto Farm (An Toàn)"
        end
    end)
    
    return {
        Notify = function(t, c)
            print("["..t.."]: "..c)
        end
    }
end

local Win = Kavo:CreateWindow("Blox Fruits Kavo Menu (Offline UI)", "DarkTheme")
