-- CUSTOM OFFLINE KAVO UI IMPLEMENTATION
local Kavo = {}
Kavo.Tabs = {}

function Kavo:CreateWindow(title, theme)
    local ScreenGui = Instance.new("ScreenGui", game:GetService("CoreGui"))
    ScreenGui.Name = "KavoUI"
    
    local MainFrame = Instance.new("Frame", ScreenGui)
    MainFrame.Size = UDim2.new(0, 550, 0, 400)
    MainFrame.Position = UDim2.new(0.5, -275, 0.5, -200)
    MainFrame.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
    MainFrame.Active = true
    MainFrame.Draggable = true
    
    local TopBar = Instance.new("Frame", MainFrame)
    TopBar.Size = UDim2.new(1, 0, 0, 40)
    TopBar.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    
    local TitleLabel = Instance.new("TextLabel", TopBar)
    TitleLabel.Size = UDim2.new(1, -20, 1, 0)
    TitleLabel.Position = UDim2.new(0, 10, 0, 0)
    TitleLabel.Text = title
    TitleLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    TitleLabel.TextSize = 18
    TitleLabel.Font = Enum.Font.SourceSansBold
    TitleLabel.TextXAlignment = Enum.TextXAlignment.Left
    
    local TabContainer = Instance.new("Frame", MainFrame)
    TabContainer.Size = UDim2.new(0, 140, 1, -40)
    TabContainer.Position = UDim2.new(0, 0, 0, 40)
    TabContainer.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
    
    local TabListLayout = Instance.new("UIListLayout", TabContainer)
    TabListLayout.SortOrder = Enum.SortOrder.LayoutOrder
    
    local ContentContainer = Instance.new("Frame", MainFrame)
    ContentContainer.Size = UDim2.new(1, -140, 1, -40)
    ContentContainer.Position = UDim2.new(0, 140, 0, 40)
    ContentContainer.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
    
    -- Floating Toggle Button
    local LogoBtn = Instance.new("ImageButton", ScreenGui)
    LogoBtn.Size = UDim2.new(0, 50, 0, 50)
    LogoBtn.Position = UDim2.new(0, 15, 0.4, 0)
    LogoBtn.Image = "rbxassetid://6031075931"
    LogoBtn.Draggable = true
    LogoBtn.Active = true
    LogoBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
    LogoBtn.BorderSizePixel = 2
    LogoBtn.BorderColor3 = Color3.fromRGB(0, 170, 255)
    local Corner = Instance.new("UICorner", LogoBtn)
    Corner.CornerRadius = UDim.new(1, 0)
    
    LogoBtn.MouseButton1Click:Connect(function()
        MainFrame.Visible = not MainFrame.Visible
    end)
    
    local WindowObj = {}
    
    function WindowObj:NewTab(tabName)
        local TabButton = Instance.new("TextButton", TabContainer)
        TabButton.Size = UDim2.new(1, 0, 0, 40)
        TabButton.Text = tabName
        TabButton.TextColor3 = Color3.fromRGB(180, 180, 180)
        TabButton.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
        TabButton.Font = Enum.Font.SourceSans
        TabButton.TextSize = 14
        
        local TabContent = Instance.new("ScrollingFrame", ContentContainer)
        TabContent.Size = UDim2.new(1, 0, 1, 0)
        TabContent.BackgroundTransparency = 1
        TabContent.Visible = false
        TabContent.CanvasSize = UDim2.new(0, 0, 2, 0)
        
        local ContentLayout = Instance.new("UIListLayout", TabContent)
        ContentLayout.SortOrder = Enum.SortOrder.LayoutOrder
        ContentLayout.Padding = UDim.new(0, 5)
        
        TabButton.MouseButton1Click:Connect(function()
            for _, child in pairs(ContentContainer:GetChildren()) do
                if child:IsA("ScrollingFrame") then child.Visible = false end
            end
            for _, btn in pairs(TabContainer:GetChildren()) do
                if btn:IsA("TextButton") then btn.TextColor3 = Color3.fromRGB(180, 180, 180) end
            end
            TabContent.Visible = true
            TabButton.TextColor3 = Color3.fromRGB(0, 170, 255)
        end)
        
        if #TabContainer:GetChildren() == 2 then
            TabContent.Visible = true
            TabButton.TextColor3 = Color3.fromRGB(0, 170, 255)
        end
        
        local TabObj = {}
        
        function TabObj:NewToggle(text, callback)
            local ToggleFrame = Instance.new("Frame", TabContent)
            ToggleFrame.Size = UDim2.new(1, -10, 0, 35)
            ToggleFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
            
            local Label = Instance.new("TextLabel", ToggleFrame)
            Label.Size = UDim2.new(0.7, 0, 1, 0)
            Label.Position = UDim2.new(0, 10, 0, 0)
            Label.Text = text
            Label.TextColor3 = Color3.fromRGB(255, 255, 255)
            Label.TextXAlignment = Enum.TextXAlignment.Left
            Label.BackgroundTransparency = 1
            
            local Btn = Instance.new("TextButton", ToggleFrame)
            Btn.Size = UDim2.new(0, 60, 0, 25)
            Btn.Position = UDim2.new(1, -70, 0, 5)
            Btn.Text = "OFF"
            Btn.BackgroundColor3 = Color3.fromRGB(150, 50, 50)
            Btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            
            local state = false
            Btn.MouseButton1Click:Connect(function()
                state = not state
                Btn.Text = state and "ON" or "OFF"
                Btn.BackgroundColor3 = state and Color3.fromRGB(50, 150, 50) or Color3.fromRGB(150, 50, 50)
                callback(state)
            end)
        end
        
        function TabObj:NewButton(text, callback)
            local BtnFrame = Instance.new("Frame", TabContent)
            BtnFrame.Size = UDim2.new(1, -10, 0, 35)
            BtnFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
            
            local Btn = Instance.new("TextButton", BtnFrame)
            Btn.Size = UDim2.new(1, -20, 1, -10)
            Btn.Position = UDim2.new(0, 10, 0, 5)
            Btn.Text = text
            Btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
            Btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            Btn.MouseButton1Click:Connect(callback)
        end
        
        return TabObj
    end
    
    return WindowObj
end

-- Auto Quest Logic Selector
local function GetCurrentQuest()
    local myLevel = LocalPlayer.Data.Level.Value
    local seaData = QuestData[game.PlaceId] or QuestData[2753915549]
    for _, quest in pairs(seaData) do
        if myLevel >= quest.MinLvl and myLevel <= quest.MaxLvl then
            return quest
        end
    end
    return nil
end

-- Initialize Window
local win = Kavo:CreateWindow("Blox Fruits Auto Quest Menu", "Dark")
local tab1 = win:NewTab("Auto Farm")
local tab2 = win:NewTab("Teleports")
local tab3 = win:NewTab("Sea Travel")

tab1:NewToggle("Auto Farm Level + Auto Quest", function(state)
    Flags.AutoFarm = state
    if state then
        task.spawn(function()
            while Flags.AutoFarm do
                task.wait(1)
                local currentQuest = GetCurrentQuest()
                if currentQuest then
                    -- Client remote logic template to take quest smoothly
                    print("Taking Quest: " .. currentQuest.QuestName)
                end
            end
        end)
    end
end)

tab1:NewToggle("Auto Bring Mobs", function(state)
    Flags.BringMob = state
end)

tab2:NewButton("Tween to Current Quest NPC", function()
    local quest = GetCurrentQuest()
    if quest and workspace:FindFirstChild(quest.NPC) then
        print("Tweening to NPC: " .. quest.NPC)
    end
end)

tab3:NewButton("Teleport to Sea 1", function() TeleportService:Teleport(2753915549, LocalPlayer) end)
